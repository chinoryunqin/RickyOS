    RICKYOS_PRODUCT=1 CROSSPOINT_PRODUCT_NAME="RickyOS" CROSSPOINT_VERSION="${RICKYOS_PLUS_APP_VERSION}"
    CROSSMUX_UI_PROFILE_HIGH_DPI=1 HYPHENATION_ENGLISH_ONLY=1
    XML_GE=0 XML_CONTEXT_BYTES=1024 MINIZ_NO_ZLIB_COMPATIBLE_NAMES=1
    PNG_MAX_BUFFERED_PIXELS=16416
    FREEINK_FONT_ENABLE_AUTOHINT=1 FREEINK_NET_WOLFSSL=1 FREEINK_CONTENT_WOLFSSL=1
    CONTENT_EXTERNAL_MINIZ=1 WOLFSSL_KEY_GEN=1 WC_RC2=1
    NO_WOLFSSL_ESP32_CRYPT_RSA_PRI=1 WOLFSSL_USER_SETTINGS=1 WOLFSSL_OPTIONS_H=1
    WOLFSSL_CLIENT_EXAMPLE=1 WOLFSSL_TLS13=1 WOLFSSL_HAVE_SP_ECC=1
    WOLFSSL_HAVE_SP_RSA=1 WOLFSSL_SP_SMALL=1 WOLFSSL_SP_4096=1
    HAVE_MAX_FRAGMENT=1 HAVE_FFDHE_2048=1 HAVE_CURVE25519=1 HAVE_SNI=1
    ENABLE_CHINESE_VERSION=1)
target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-missing-field-initializers -Wno-error)

set_target_properties(${COMPONENT_LIB} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)

# QRCode 0.0.1 supplies its own C bool typedef; compile only this legacy unit as C11.
set_source_files_properties("${DEPENDENCY_ROOT}/QRCode/src/qrcode.c" PROPERTIES COMPILE_OPTIONS "-std=gnu11")

# Preserve the complete 4-bit UI with headroom in the 6.25 MiB app slot.
# Limit LTO to application/component sources; IDF and radio drivers remain separate.
target_compile_options(${COMPONENT_LIB} PRIVATE -flto=auto)
idf_build_set_property(LINK_OPTIONS "-flto=auto;-fuse-linker-plugin" APPEND)
# Keep all applications and glyphs when USB is enabled; GCC 15 size optimization.
# Hardware timing and radio/IDF driver compilation are unchanged.
target_compile_options(${COMPONENT_LIB} PRIVATE -Oz)

# Explicitly opt in to unattended diagnostics; no flag is enabled in normal builds.
foreach(TEST_OPTION NETTEST NETTEST_CONNECTED NETTEST_AX NETTEST_HEADLESS NETTEST_NATIVE NETTEST_BG NETTEST_CONFIG_BUFFERS)
  if("$ENV{RICKYOS_PLUS_${TEST_OPTION}}" STREQUAL "1")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE RICKYOS_PLUS_${TEST_OPTION}=1)
  endif()
endforeach()
if("$ENV{RICKYOS_PLUS_NETTEST_NATIVE}" STREQUAL "1" AND
   NOT "$ENV{RICKYOS_PLUS_NETTEST_HEADLESS}" STREQUAL "1")
  message(FATAL_ERROR "Native Wi-Fi isolation requires RICKYOS_PLUS_NETTEST_HEADLESS=1")
endif()

if("$ENV{RICKYOS_PLUS_NETTEST_CONNECTED}" STREQUAL "1")
  if(NOT "$ENV{RICKYOS_PLUS_NETTEST}" STREQUAL "1" OR "$ENV{RICKYOS_PLUS_NETTEST_HEADLESS}" STREQUAL "1")
    message(FATAL_ERROR "Connected comparison requires NETTEST=1 and normal application UI")
  endif()
endif()
